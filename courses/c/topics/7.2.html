7.2 Array Operations in C

What Are Array Operations?

Array operations are tasks performed on array elements using loops and index manipulation. These operations help you add, remove, update, and search elements in an array.

Common Array Operations

  • Traversal: Visit each element
  • Insertion: Add a new element
  • Deletion: Remove an element
  • Searching: Find an element
  • Updating: Modify an element

Example: Traversing an Array

#include <stdio.h>

int main() {
    int arr[5] = {1, 2, 3, 4, 5};

    printf("Array elements: ");
    for (int i = 0; i < 5; i++) {
        printf("%d ", arr[i]);
    }

    return 0;
}

Example: Insert Element at End

#include <stdio.h>

int main() {
    int arr[6] = {10, 20, 30, 40, 50};
    int size = 5;
    int newElement = 60;

    arr[size] = newElement;
    size++;

    printf("Updated array: ");
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }

    return 0;
}

Example: Searching an Element

#include <stdio.h>

int main() {
    int arr[5] = {5, 10, 15, 20, 25};
    int key = 20;
    int found = 0;

    for (int i = 0; i < 5; i++) {
        if (arr[i] == key) {
            printf("Element %d found at index %d\\n", key, i);
            found = 1;
            break;
        }
    }

    if (!found)
        printf("Element not found.");

    return 0;
}

Example: Delete Element

#include <stdio.h>

int main() {
    int arr[6] = {10, 20, 30, 40, 50};
    int size = 5;
    int pos = 2;   // delete element at index 2 (value 30)

    for (int i = pos; i < size - 1; i++) {
        arr[i] = arr[i + 1];  // shift left
    }
    size--;

    printf("Array after deletion: ");
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }

    return 0;
}

Summary Table

Operation Description

Insertion Add a new element at a specific position
