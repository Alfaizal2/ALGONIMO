10.3 fprintf(), fscanf(), fgets()

Explanation

These three functions are commonly used for formatted reading and writing in files:

  • fprintf() – formatted writing (like printf())
  • fscanf() – formatted reading (like scanf())
  • fgets() – read one full line from a file

fprintf()

fprintf() is used to write formatted text into a file.

#include <stdio.h>

int main() {
    FILE *fp = fopen("data.txt", "w");

    if (fp != NULL) {
        int age = 25;
        fprintf(fp, "Age: %d\n", age);
        fclose(fp);
    }
    return 0;
}
Output:
Creates data.txt with:
Age: 25

fscanf()

fscanf() reads formatted data from a file.

#include <stdio.h>

int main() {
    FILE *fp = fopen("data.txt", "r");

    if (fp != NULL) {
        int age;
        fscanf(fp, "Age: %d", &age);
        printf("Read from file: %d\n", age);
        fclose(fp);
    }
    return 0;
}
Output: Read from file: 25

fgets()

fgets() reads a full line (string) from a file.

#include <stdio.h>

int main() {
    char str[100];
    FILE *fp = fopen("data.txt", "r");

    if (fp != NULL) {
        fgets(str, sizeof(str), fp);
        printf("Line: %s", str);
        fclose(fp);
    }
    return 0;
}
Output:
Line: Age: 25
